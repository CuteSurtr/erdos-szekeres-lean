import Lean
import EncB
import EncB.Emit
open Lean Elab Command

/-! Every constant declared in a module `EncB.*` (the package's own declarations, from the imported .olean files),
with its module, the line of its name (`-` if Lean generated it), its kind and `#print axioms` set (`Lean.collectAxioms`, the function behind
`#print axioms`). Compiler- and equation-compiler auxiliaries (`Name.isInternalDetail`, `match_`/`proof_`/`eq_`
etc.) are counted but listed only as a total; their axioms flow into the listed declarations that use them. -/

def stdAxioms : List Name := [``propext, ``Classical.choice, ``Quot.sound]

#eval show CommandElabM Unit from do
  let env ← getEnv
  let mods := env.header.moduleNames
  let mut rows : Array (Name × Nat × String × Name × List Name) := #[]
  let mut internal := 0
  let mut internalBad : Array Name := #[]
  for (n, ci) in env.constants.map₁.toList do
    let some idx := env.getModuleIdxFor? n | continue
    let mod := mods[idx.toNat]!
    unless (`EncB).isPrefixOf mod do continue
    let axs := ((← liftCoreM (collectAxioms n)).qsort Name.lt).toList
    if n.isInternalDetail || isAuxRecursor env n || isNoConfusion env n then
      internal := internal + 1
      unless axs.all stdAxioms.contains do internalBad := internalBad.push n
      continue
    let kind := match ci with
      | .thmInfo _ => "theorem" | .defnInfo _ => "def" | .inductInfo _ => "inductive"
      | .ctorInfo _ => "ctor" | .recInfo _ => "rec" | .axiomInfo _ => "AXIOM" | .opaqueInfo _ => "opaque"
      | .quotInfo _ => "quot"
    let line ← match (← liftCoreM (findDeclarationRanges? n)) with
      | some r => pure r.selectionRange.pos.line
      | none => pure 0
    rows := rows.push (mod, line, kind, n, axs)
  rows := rows.qsort fun a b => a.1.toString < b.1.toString || (a.1 == b.1 && (a.2.1 < b.2.1 ||
    (a.2.1 == b.2.1 && a.2.2.2.1.toString < b.2.2.2.1.toString)))
  let mut out := ""
  let mut nThm := 0
  let mut nThmSrc := 0
  let mut bad : Array Name := #[]
  for (mod, line, kind, n, axs) in rows do
    if kind == "theorem" then nThm := nThm + 1
    if kind == "theorem" && line != 0 then nThmSrc := nThmSrc + 1
    unless axs.all stdAxioms.contains do bad := bad.push n
    let axStr := if axs.isEmpty then "none" else ", ".intercalate (axs.map toString)
    let lineStr := if line == 0 then "-" else toString line
    out := out ++ s!"{mod}:{lineStr}\t{kind}\t{n}\t[{axStr}]\n"
  out := out ++ s!"SUMMARY: {rows.size} named declarations in EncB.* ({nThm} theorems, {nThmSrc} of them written in the source; line '-' = generated by Lean, e.g. injEq), {internal} internal auxiliaries; " ++
    s!"outside {stdAxioms}: {bad.size} named {bad.toList}, {internalBad.size} internal {internalBad.toList}\n"
  logInfo out
